 place(){
  const g=this.g,t=this.t;
  if(!t.boat&&!t.aircraft){this.pitch=0;this.roll=0;this.pV=0;this.rV=0;}

  this.v.ch.position.y=lift;
 }
}

function stepAircraft(c,dt){

 // Build speed on the runway, then press Space once to lift off.
