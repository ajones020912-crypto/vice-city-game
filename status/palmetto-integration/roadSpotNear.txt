 for(const r of roads){if(r.type==='hwy')continue;const along=r.o==='h'?x:z,lat=r.o==='h'?z:x;const p=clamp(along,r.a+12,r.b-12),side=lat>=r.c?1:-1,L=r.c+side*(r.hw-1.4);
