function stepCar(c,dt){
 const t=c.t,inp=c.inp;if(t.aircraft)return stepAircraft(c,dt);if(t.boat)return stepBoat(c,dt);c.age+=dt;const up=c.up||{},pw=1+.15*(up.eng||0)+.18*(up.turbo||0),tw=1+.07*(up.eng||0);
 if(c.dead){inp.thr=0;inp.hb=0;inp.steer=0}
 // fuel
 if(c.fuel==null)c.fuel=80;
 if(c.occ==='player'||c.mode==='player'){
  if(Math.abs(inp.thr)>0.05&&c.speed>0.3)c.fuel=Math.max(0,c.fuel-dt*(0.35+c.speed*0.012)*(1-.3*(up.tank||0)));
  if(c.fuel<=0){inp.thr=0;if(c===pl.car&&!(G.fuelWarn>G.time)){toast('Out of fuel! Push on to a ⛽ gas station or call Scooter Bros',2600);G.fuelWarn=G.time+6}}
 }
 let sn=Math.sin(c.h),cs=Math.cos(c.h),vf=c.vx*sn+c.vz*cs,vl=c.vx*cs-c.vz*sn,a=0;
 const water=WL-heightAt(c.x,c.z);
 if(water>1.6&&!c.dead){c.drown+=dt;if(c.drown>2.2)damageCar(c,999,'water')}
 const wetDrag=water>.15?1+Math.min(water,1.5)*1.6:1;
 if(inp.thr>0){if(vf<-.4)a=24;else a=(t.power*pw*inp.thr*Math.max(0,1-Math.pow(Math.max(vf,0)/(t.top*tw),2.3))+(vf<6?2.2*inp.thr:0))/wetDrag}
 else if(inp.thr<0){if(vf>.4)a=-(27*(1+.22*(up.brk||0)))*-inp.thr;else a=-(7*-inp.thr)*Math.max(0,1-(-vf)/11)}
 else if(Math.abs(vf)>.01){a=-Math.sign(vf)*Math.min(Math.abs(vf)/dt,1.4)}
 c.braking=(inp.thr<0&&vf>.4)||(inp.thr>0&&vf<-.4)||inp.hb>0;
 if(inp.hb&&Math.abs(vf)>.01)a-=Math.sign(vf)*Math.min(Math.abs(vf)/dt,11*(1+.15*(up.brk||0)));
 a-=vf*Math.abs(vf)*.0012*(1);a-=vf*(wetDrag-1)*.9;
 const gF=(heightAt(c.x+sn*t.wb/2,c.z+cs*t.wb/2)-heightAt(c.x-sn*t.wb/2,c.z-cs*t.wb/2))/t.wb;a-=9.8*gF*.8;
 vf+=a*dt;c.lastAcc=a;
 const slip=Math.abs(vl)/(Math.abs(vf)+3);let gr=t.grip*(1+.18*(up.grip||0))*(inp.hb&&Math.abs(vf)>3?.13:1)/(1+slip*1.1)/(wetDrag>1?1.3:1);
 vl*=Math.exp(-gr*dt);
 const maxS=t.steer*(1+.12*(up.grip||0))/(1+Math.abs(vf)*.08);c.steer+=(inp.steer*maxS-c.steer)*Math.min(1,dt*7);
 let yrT=vf*Math.tan(c.steer)/t.wb;if(inp.hb&&Math.abs(vf)>3)yrT*=1.7;yrT=clamp(yrT,-2.6,2.6);
 c.yr+=(yrT-c.yr)*Math.min(1,dt*(inp.hb?5:11));
 c.vx=vf*sn+vl*cs;c.vz=vf*cs-vl*sn;c.vf=vf;c.vl=vl;
 c.x+=c.vx*dt;c.z+=c.vz*dt;c.h+=c.yr*dt;
 // ground & suspension — sample under all four wheels + center
 sn=Math.sin(c.h);cs=Math.cos(c.h);const wb=t.wb/2,hw=t.W/2;
 const hC=heightAt(c.x,c.z);
 const hFL=heightAt(c.x+sn*wb+cs*hw,c.z+cs*wb-sn*hw),hFR=heightAt(c.x+sn*wb-cs*hw,c.z+cs*wb+sn*hw);
 const hBL=heightAt(c.x-sn*wb+cs*hw,c.z-cs*wb-sn*hw),hBR=heightAt(c.x-sn*wb-cs*hw,c.z-cs*wb+sn*hw);
 const hF=(hFL+hFR)*.5,hB=(hBL+hBR)*.5,hL=(hFL+hBL)*.5,hR=(hFR+hBR)*.5;
 // ride height: keep car on the highest supporting ground under the chassis
 const targetY=Math.max(hC,hF,hB,hL,hR,(hFL+hFR+hBL+hBR)*.25);
 // snap hard when below ground, smooth when above (prevents tunneling into slopes)
 if(c.y<targetY-.02)c.y=targetY;else c.y=lerp(c.y,targetY,Math.min(1,dt*22));
 // pitch / roll from wheel heights, clamped so cars never tip into the mesh
 let pT=-Math.atan2(hF-hB,t.wb)-.011*c.lastAcc;
 let rT=Math.atan2(hL-hR,t.W)+.013*vf*c.yr;
 pT=clamp(pT,-.38,.38);rT=clamp(rT,-.32,.32);
 c.pV+=((pT-c.pitch)*140-c.pV*14)*dt;c.rV+=((rT-c.roll)*140-c.rV*14)*dt;
 c.pitch=clamp(c.pitch+c.pV*dt,-.4,.4);c.roll=clamp(c.roll+c.rV*dt,-.35,.35);
 c.wheelRot+=vf/t.wr*dt;
 // static collisions
 carCircles(c,_cc);const r=hw+.08;let hit=0,crumpleContact=null;
 for(let k=0;k<3;k++){const p=_cc[k];const cols=queryCols(p[0],p[1],r+1);
  for(const col of cols){const ct=circleBox(p,r,col);if(!ct)continue;
   c.x+=ct.nx*ct.pen;c.z+=ct.nz*ct.pen;p[0]+=ct.nx*ct.pen;p[1]+=ct.nz*ct.pen;
   const vn=c.vx*ct.nx+c.vz*ct.nz;if(vn<0){c.vx-=(1.3)*vn*ct.nx;c.vz-=(1.3)*vn*ct.nz;c.vx*=.985;c.vz*=.985;const imp=-vn;if(imp>hit)crumpleContact={x:p[0]-ct.nx*r,z:p[1]-ct.nz*r,nx:ct.nx,nz:ct.nz};hit=Math.max(hit,imp);
    const nl=ct.nx*cs-ct.nz*sn,off=(k-1)*-(t.L/2-.95);c.yr+=off*nl*imp*.035;}}}
 if(hit>1)crumpleCrash(c,hit,crumpleContact);if(hit>2.5){carImpact(c,hit)}
 // world bounds
 c.x=clamp(c.x,-WS+150,WS-150);c.z=clamp(c.z,-WS+150,WS-150);
 // tire smoke / drift
 if(!c.dead&&Math.abs(vl)>3.2&&Math.abs(vf)>4&&Math.random()<dt*28&&water<=.1){emit(c.x-sn*1.4,c.y+.2,c.z-cs*1.4,(Math.random()-.5)*1.5,.6,(Math.random()-.5)*1.5,1.1,.5,2,DUST,DUST,-.2,.5)}
 if(c.hp<40&&!c.dead&&Math.random()<dt*10*(1-c.hp/40)){emit(c.x+sn*1.8,c.y+1,c.z+cs*1.8,0,1.5,0,1.6,.4,1.8,SMK,SMK2,-.6,.6)}
 if(c.fire>0){c.fire-=dt;emit(c.x+(Math.random()-.5)*1.5,c.y+.8,c.z+(Math.random()-.5)*1.5,0,2+Math.random()*2,0,.6,.7,.2,FIRE,FIRE2,-1,.8);if(Math.random()<dt*8)emit(c.x,c.y+1.4,c.z,0,2,0,2,.8,2.5,SMK,SMK2,-.6,.5)}
}
