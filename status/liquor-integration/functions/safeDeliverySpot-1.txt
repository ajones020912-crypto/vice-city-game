function safeDeliverySpot(c){for(let r=12;r<=60;r+=8)for(let i=0;i<12;i++){const a=i/12*TAU,rs=roadSpotNear(pl.x+Math.cos(a)*r,pl.z+Math.sin(a)*r);if(!rs||Math.hypot(rs.x-pl.x,rs.z-pl.z)<8||Math.hypot(rs.x-pl.x,rs.z-pl.z)>100||heightAt(rs.x,rs.z)<WL+.2)continue;if(cars.some(v=>v!==c&&Math.hypot(v.x-rs.x,v.z-rs.z)<7))continue;if(queryCols(rs.x,rs.z,4).some(b=>circleBox([rs.x,rs.z],3,b)))continue;return rs}return null}
